   116→function FilterableCard({ category, title, searchQuery, activeFilter, children }: { category: string; title: string; searchQuery: string; activeFilter: string; children: React.ReactNode }) {
   117→  const matchesCategory = activeFilter === 'All' || category === activeFilter;
   118→  const q = searchQuery.toLowerCase();
   119→  const matchesSearch = q === '' || title.toLowerCase().includes(q);
   120→  if (!matchesCategory || !matchesSearch) return null;
   121→  return <>{children}</>;
   122→}
