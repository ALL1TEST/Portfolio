     4→import { motion } from 'framer-motion';
     5→import { toast } from 'sonner';
     6→import { Save, Loader2, User, Upload, Trash2, Camera, FileText } from 'lucide-react';
     7→import { Button } from '@/components/ui/button';
     8→import { Input } from '@/components/ui/input';
     9→import { Label } from '@/components/ui/label';

    11→import { Card } from '@/components/ui/card';

    62→  fullName: '',
    63→  brandName: '',
    64→  professionalTitle: '',
    65→  shortBio: '',
    66→  aboutText: '',
    67→  footerBio: '',
    68→  email: '',
    69→  phone: '',
    70→  location: '',
    71→  githubUrl: '',
    72→  linkedinUrl: '',
    73→  instagramUrl: '',
    74→  twitterUrl: '',
    75→  featuredProjectsTitle: '',

    77→  aboutCard1Title: '',
    78→  aboutCard1Description: '',
    79→  aboutCard2Title: '',
    80→  aboutCard2Description: '',
    81→  aboutCard3Title: '',
    82→  aboutCard3Description: '',
    83→  logoUrl: '',
    84→  profileImage: '',
    85→  cvFile: '',
    86→  stat1Value: '',

    98→  resumeIntro: '',

   100→  resumeExpTitle: '',
   101→  contactPageTitle: '',

   106→
   107→export default function SettingsPage() {
   108→  const [profile, setProfile] = useState<Profile>(emptyProfile);
   109→  const [original, setOriginal] = useState<Profile>(emptyProfile);
   110→  const [loading, setLoading] = useState(true);
   111→  const [saving, setSaving] = useState(false);
   112→  const [uploading, setUploading] = useState<'profile' | 'cv' | 'logo' | null>(null);
   113→  const imageInputRef = useRef<HTMLInputElement>(null);
   114→  const cvInputRef = useRef<HTMLInputElement>(null);
   115→  const logoInputRef = useRef<HTMLInputElement>(null);
   116→
   117→  useEffect(() => {
   118→    const fetchProfile = async () => {
   119→      try {
   120→        const res = await fetch('/api/profile');
   121→        if (res.ok) {
   122→          const data = await res.json();
   123→          setProfile(data);
   124→          setOriginal(data);
   125→        }
   126→      } catch {
   127→        toast.error('Failed to load profile');
   128→      } finally {
   129→        setLoading(false);
   130→      }
   131→    };
   132→    fetchProfile();
   133→  }, []);
   134→
   135→  const updateField = (field: keyof Profile, value: string) => {
   136→    setProfile((prev) => ({ ...prev, [field]: value }));
   137→  };
   138→
   139→  const handleImageUpload = async (e: React.ChangeEvent<HTMLInputElement>) => {
   140→    const file = e.target.files?.[0];
   141→    if (!file) return;
   142→    if (!file.type.startsWith('image/')) { toast.error('Only image files are allowed'); return; }
   143→    if (file.size > 5 * 1024 * 1024) { toast.error('Image must be under 5MB'); return; }
   144→
   145→    setUploading('profile');
   146→    try {
   147→      const formData = new FormData();
   148→      formData.append('file', file);
   149→      formData.append('category', 'profile');
   150→      const res = await fetch('/api/upload', { method: 'POST', body: formData });
   151→      if (res.ok) {
   152→        const data = await res.json();
   153→        updateField('profileImage', data.url);
   154→        toast.success('Image uploaded successfully');
   155→      } else { toast.error('Failed to upload image'); }
   156→    } catch { toast.error('Failed to upload image'); }
   157→    finally { setUploading(null); if (imageInputRef.current) imageInputRef.current.value = ''; }
   158→  };
   159→
   160→  const handleCvUpload = async (e: React.ChangeEvent<HTMLInputElement>) => {
   161→    const file = e.target.files?.[0];
   162→    if (!file) return;
   163→    if (file.type !== 'application/pdf') { toast.error('Only PDF files are allowed'); return; }
   164→    if (file.size > 10 * 1024 * 1024) { toast.error('File must be under 10MB'); return; }
   165→
   166→    setUploading('cv');
   167→    try {
   168→      const formData = new FormData();
   169→      formData.append('file', file);
   170→      formData.append('category', 'cv');
   171→      const res = await fetch('/api/upload', { method: 'POST', body: formData });
   172→      if (res.ok) {
   173→        const data = await res.json();
   174→        updateField('cvFile', data.url);
   175→        toast.success('CV uploaded successfully');
   176→      } else { toast.error('Failed to upload CV'); }
   177→    } catch { toast.error('Failed to upload CV'); }
   178→    finally { setUploading(null); if (cvInputRef.current) cvInputRef.current.value = ''; }
   179→  };
   180→
   181→  const handleLogoUpload = async (e: React.ChangeEvent<HTMLInputElement>) => {
   182→    const file = e.target.files?.[0];
   183→    if (!file) return;
   184→    if (!file.type.startsWith('image/')) { toast.error('Only image files are allowed'); return; }
   185→    if (file.size > 5 * 1024 * 1024) { toast.error('Image must be under 5MB'); return; }
   186→
   187→    setUploading('logo');
   188→    try {
   189→      const formData = new FormData();
   190→      formData.append('file', file);
   191→      formData.append('category', 'profile');
   192→      const res = await fetch('/api/upload', { method: 'POST', body: formData });
   193→      if (res.ok) {
   194→        const data = await res.json();
   195→        updateField('logoUrl', data.url);
   196→        toast.success('Logo uploaded successfully');
   197→      } else { toast.error('Failed to upload logo'); }
   198→    } catch { toast.error('Failed to upload logo'); }
   199→    finally { setUploading(null); if (logoInputRef.current) logoInputRef.current.value = ''; }
   200→  };
   201→
   202→  const handleSave = async () => {
   203→    setSaving(true);
   204→    try {
   205→      const res = await fetch('/api/profile', {
   206→        method: 'PUT',
   207→        headers: { 'Content-Type': 'application/json' },
   208→        body: JSON.stringify(profile),
   209→      });
   210→      if (res.ok) {
   211→        const data = await res.json();
   212→        setOriginal(data);
   213→        toast.success('Profile saved successfully');
   214→      } else { toast.error('Failed to save profile'); }
   215→    } catch { toast.error('Failed to save profile'); }
   216→    finally { setSaving(false); }
   217→  };
   218→
   219→  const hasChanges = JSON.stringify(profile) !== JSON.stringify(original);
   220→
   221→  if (loading) {
   222→    return (
   223→      <div className="space-y-4">
   224→        {[1, 2, 3, 4, 5].map((i) => (
   225→          <Skeleton key={i} className="h-12 w-full bg-surface rounded-xl" />
   226→        ))}
   227→      </div>
   228→    );
   229→  }
   230→
   231→  return (
   232→    <motion.div
   233→      initial={{ opacity: 0, y: 20 }}
   234→      animate={{ opacity: 1, y: 0 }}
   235→      transition={{ duration: 0.4 }}
   236→      className="space-y-6 max-w-3xl"
   237→    >
   238→      {/* Header with save button */}
   239→      <div className="flex items-center justify-between">
   240→        <div className="flex items-center gap-3">
   241→          <div className="w-10 h-10 rounded-lg bg-brand/10 flex items-center justify-center">
   242→            <User className="w-5 h-5 text-brand" />
   243→          </div>
   244→          <div>
   245→            <h2 className="text-base font-semibold text-white">Profile Information</h2>
   246→            <p className="text-sm text-muted-text">Update your personal and professional details</p>
   247→          </div>
   248→        </div>
   249→        <Button
   250→          onClick={handleSave}
   251→          disabled={saving || !hasChanges}
   252→          className="bg-brand hover:bg-brand-light text-white gap-2 disabled:opacity-40"
   253→        >
   254→          {saving ? <Loader2 className="w-4 h-4 animate-spin" /> : <Save className="w-4 h-4" />}
   255→          Save Changes
   256→        </Button>
   257→      </div>
   258→
   259→      {/* Profile Image */}
   260→      <Card className="bg-surface border-stroke p-6">
   261→        <h3 className="text-sm font-semibold text-white mb-4">Profile Image</h3>
   262→        <div className="flex items-center gap-6">
   263→          <div className="relative w-24 h-24 rounded-xl overflow-hidden border-2 border-stroke/50 bg-dark flex-shrink-0">
   264→            {profile.profileImage ? (
   265→              <img src={profile.profileImage} alt="Profile" className="w-full h-full object-cover" />
   266→            ) : (
   267→              <div className="w-full h-full flex items-center justify-center text-muted-text"><User className="w-8 h-8" /></div>
   268→            )}
   269→          </div>
   270→          <div className="flex-1 space-y-3">
   271→            <div className="flex items-center gap-2">
   272→              <Button type="button" variant="outline" size="sm" onClick={() => imageInputRef.current?.click()} disabled={uploading === 'profile'} className="border-stroke text-white hover:bg-dark gap-2">
   273→                {uploading === 'profile' ? <Loader2 className="w-4 h-4 animate-spin" /> : <Camera className="w-4 h-4" />}
   274→                {uploading === 'profile' ? 'Uploading...' : 'Upload Image'}
   275→              </Button>
   276→              {profile.profileImage && (
   277→                <Button type="button" variant="ghost" size="sm" onClick={() => updateField('profileImage', '')} className="text-red-400 hover:text-red-300 hover:bg-red-400/10 gap-1">
   278→                  <Trash2 className="w-3.5 h-3.5" /> Remove
   279→                </Button>
   280→              )}
   281→            </div>
   282→            <p className="text-xs text-muted-text">Recommended: Square image, min 300×300px. Max 5MB.</p>
   283→            <input ref={imageInputRef} type="file" accept="image/*" onChange={handleImageUpload} className="hidden" />
   284→          </div>
   285→        </div>
   286→      </Card>
   287→
   288→      {/* CV Upload */}
   289→      <Card className="bg-surface border-stroke p-6">
   290→        <h3 className="text-sm font-semibold text-white mb-4">CV / Resume</h3>
   291→        <div className="flex items-center gap-6">
   292→          <div className="relative w-24 h-24 rounded-xl overflow-hidden border-2 border-stroke/50 bg-dark flex-shrink-0 flex items-center justify-center">
   293→            {profile.cvFile ? (
   294→              <a href={profile.cvFile} target="_blank" rel="noopener noreferrer" className="flex flex-col items-center gap-1 text-muted-text hover:text-brand transition-colors">
   295→                <FileText className="w-8 h-8" />
   296→                <span className="text-[10px]">View CV</span>
   297→              </a>
   298→            ) : (
   299→              <div className="flex flex-col items-center gap-1 text-muted-text/40">
   300→                <FileText className="w-8 h-8" />
   301→                <span className="text-[10px]">No CV</span>
   302→              </div>
   303→            )}
   304→          </div>
   305→          <div className="flex-1 space-y-3">
   306→            <div className="flex items-center gap-2">
   307→              <Button type="button" variant="outline" size="sm" onClick={() => cvInputRef.current?.click()} disabled={uploading === 'cv'} className="border-stroke text-white hover:bg-dark gap-2">
   308→                {uploading === 'cv' ? <Loader2 className="w-4 h-4 animate-spin" /> : <Upload className="w-4 h-4" />}
   309→                {uploading === 'cv' ? 'Uploading...' : 'Upload CV'}
   310→              </Button>
   311→              {profile.cvFile && (
   312→                <Button type="button" variant="ghost" size="sm" onClick={() => updateField('cvFile', '')} className="text-red-400 hover:text-red-300 hover:bg-red-400/10 gap-1">
   313→                  <Trash2 className="w-3.5 h-3.5" /> Remove
   314→                </Button>
   315→              )}
   316→            </div>
   317→            <p className="text-xs text-muted-text">PDF format only. Max 10MB.</p>
   318→            <input ref={cvInputRef} type="file" accept=".pdf" onChange={handleCvUpload} className="hidden" />
   319→          </div>
   320→        </div>
   321→      </Card>
   322→
   323→      {/* Logo */}
   324→      <Card className="bg-surface border-stroke p-6">
   325→        <h3 className="text-sm font-semibold text-white mb-4">Logo</h3>
   326→        <div className="flex items-center gap-6">
   327→          <div className="relative w-24 h-24 rounded-xl overflow-hidden border-2 border-stroke/50 bg-dark flex-shrink-0">
   328→            {profile.logoUrl ? (
   329→              <img src={profile.logoUrl} alt="Logo" className="w-full h-full object-contain p-1" />
   330→            ) : (
   331→              <div className="w-full h-full flex items-center justify-center text-muted-text/40">
   332→                <img src="/logo.png" alt="Current logo" className="w-12 h-12 object-contain opacity-50" />
   333→              </div>
   334→            )}
   335→          </div>
   336→          <div className="flex-1 space-y-3">
   337→            <div className="flex items-center gap-2">
   338→              <Button type="button" variant="outline" size="sm" onClick={() => logoInputRef.current?.click()} disabled={uploading === 'logo'} className="border-stroke text-white hover:bg-dark gap-2">
   339→                {uploading === 'logo' ? <Loader2 className="w-4 h-4 animate-spin" /> : <Camera className="w-4 h-4" />}
   340→                {uploading === 'logo' ? 'Uploading...' : 'Upload Logo'}
   341→              </Button>
   342→              {profile.logoUrl && (
   343→                <Button type="button" variant="ghost" size="sm" onClick={() => updateField('logoUrl', '')} className="text-red-400 hover:text-red-300 hover:bg-red-400/10 gap-1">
   344→                  <Trash2 className="w-3.5 h-3.5" /> Remove
   345→                </Button>
   346→              )}
   347→            </div>
   348→            <p className="text-xs text-muted-text">Recommended: Square PNG with transparent background, min 200×200px. Max 5MB.</p>
   349→            <input ref={logoInputRef} type="file" accept="image/*" onChange={handleLogoUpload} className="hidden" />
   350→          </div>
   351→        </div>
   352→      </Card>
   353→
   354→      {/* Personal Information */}
   355→      <Card className="bg-surface border-stroke p-6">
   356→        <h3 className="text-sm font-semibold text-white mb-4">Personal Information</h3>
   357→        <div className="space-y-4">
   358→          <div className="grid grid-cols-1 md:grid-cols-2 gap-4">
   359→            <div className="space-y-2">
   360→              <Label className="text-sm text-white">Full Name</Label>
   361→              <Input value={profile.fullName} onChange={(e) => updateField('fullName', e.target.value)} className="bg-dark border-stroke text-white placeholder:text-muted-text" placeholder="ABDELLAH AIT-SI" />
   362→            </div>
   363→            <div className="space-y-2">
   364→              <Label className="text-sm text-white">Brand Name</Label>
   365→              <Input value={profile.brandName} onChange={(e) => updateField('brandName', e.target.value)} className="bg-dark border-stroke text-white placeholder:text-muted-text" placeholder="CodeVirtox" />
   366→            </div>
   367→          </div>
   368→          <div className="space-y-2">
   369→            <Label className="text-sm text-white">Professional Title</Label>
   370→            <Input value={profile.professionalTitle} onChange={(e) => updateField('professionalTitle', e.target.value)} className="bg-dark border-stroke text-white placeholder:text-muted-text" placeholder="Full Stack Developer | AI & Automation" />
   371→          </div>
   372→          <div className="space-y-2">
   373→            <Label className="text-sm text-white">Short Bio</Label>
   374→            <Input value={profile.shortBio} onChange={(e) => updateField('shortBio', e.target.value)} className="bg-dark border-stroke text-white placeholder:text-muted-text" placeholder="I build modern web applications..." />
   375→          </div>
   376→        </div>
   377→      </Card>
   378→
   379→      {/* About Text */}
   380→      <Card className="bg-surface border-stroke p-6">
   381→        <h3 className="text-sm font-semibold text-white mb-4">About</h3>
   382→        <div className="space-y-2">
   383→          <Label className="text-sm text-white">About Text</Label>
   384→          <Textarea value={profile.aboutText} onChange={(e) => updateField('aboutText', e.target.value)} className="bg-dark border-stroke text-white placeholder:text-muted-text min-h-[200px]" placeholder="Tell visitors about yourself..." />
   385→          <p className="text-xs text-muted-text">This appears on the About section of your portfolio.</p>
   386→        </div>
   387→      </Card>
   388→
   389→      {/* About Cards */}
   390→      <Card className="bg-surface border-stroke p-6">
   391→        <h3 className="text-sm font-semibold text-white mb-4">About Cards</h3>
   392→        <p className="text-xs text-muted-text mb-4">Edit the 3 value cards displayed in the About section.</p>
   393→        <div className="space-y-6">
   394→          {(['1', '2', '3'] as const).map((num) => (
   395→            <div key={num} className="space-y-3">
   396→              <div className="flex items-center gap-2">
   397→                <span className="w-7 h-7 rounded-md bg-brand/10 text-brand font-bold text-xs flex items-center justify-center flex-shrink-0">0{num}</span>
   398→                <span className="text-xs font-medium text-muted-text">Card {num}</span>

   405→                    onChange={(e) => updateField(`aboutCard${num}Title` as keyof Profile, e.target.value)}
   406→                    className="bg-dark border-stroke text-white placeholder:text-muted-text"
   407→                    placeholder={`Card ${num} title...`}
   408→                  />
   409→                </div>

   411→                  <Label className="text-xs text-white/70">Description</Label>
   412→                  <Textarea
   413→                    value={profile[`aboutCard${num}Description` as keyof Profile] as string}
   414→                    onChange={(e) => updateField(`aboutCard${num}Description` as keyof Profile, e.target.value)}

   426→      {/* Featured Projects Section */}
