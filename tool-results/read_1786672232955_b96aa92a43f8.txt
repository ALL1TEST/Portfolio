    77→// ----- Education Tab -----
    78→const emptyEducation = { degree: '', field: '', institution: '', location: '', year: '', displayOrder: 0 };
    79→
    80→function EducationTab() {
    81→  const [items, setItems] = useState<Education[]>([]);
    82→  const [loading, setLoading] = useState(true);
    83→  const [saving, setSaving] = useState(false);
    84→  const [formOpen, setFormOpen] = useState(false);
    85→  const [deleteOpen, setDeleteOpen] = useState(false);
    86→  const [editingId, setEditingId] = useState<string | null>(null);
    87→  const [deleteId, setDeleteId] = useState<string | null>(null);
    88→  const [form, setForm] = useState(emptyEducation);
    89→
    90→  const fetchItems = useCallback(async () => {
    91→    try {
    92→      const res = await fetch('/api/education');
    93→      if (res.ok) setItems(await res.json());
    94→    } catch { /* ignore */ } finally { setLoading(false); }
    95→  }, []);
    96→
    97→  useEffect(() => { fetchItems(); }, [fetchItems]);
    98→
    99→  const openCreate = () => { setEditingId(null); setForm(emptyEducation); setFormOpen(true); };
   100→  const openEdit = (item: Education) => {
   101→    setEditingId(item.id);
   102→    setForm({ degree: item.degree, field: item.field, institution: item.institution, location: item.location, year: item.year, displayOrder: item.displayOrder });
   103→    setFormOpen(true);
   104→  };
   105→
   106→  const handleSubmit = async (e: React.FormEvent) => {
   107→    e.preventDefault(); setSaving(true);
   108→    try {
   109→      const res = await fetch('/api/education', {
   110→        method: editingId ? 'PUT' : 'POST',
   111→        headers: { 'Content-Type': 'application/json' },
   112→        body: JSON.stringify(editingId ? { ...form, id: editingId } : form),
   113→      });
   114→      if (res.ok) { toast.success(editingId ? 'Updated' : 'Created'); setFormOpen(false); fetchItems(); }
   115→      else toast.error('Failed to save');
   116→    } catch { toast.error('Failed to save'); } finally { setSaving(false); }
   117→  };
   118→
   119→  const handleDelete = async () => {
   120→    if (!deleteId) return;
   121→    try {
   122→      const res = await fetch(`/api/education?id=${deleteId}`, { method: 'DELETE' });
   123→      if (res.ok) { toast.success('Deleted'); setItems((p) => p.filter((i) => i.id !== deleteId)); }
   124→      else toast.error('Failed');
   125→    } catch { toast.error('Failed'); } finally { setDeleteOpen(false); setDeleteId(null); }
   126→  };
   127→
   128→  return (
   129→    <div className="space-y-3">
   130→      <div className="flex justify-end">
   131→        <Button onClick={openCreate} size="sm" className="bg-brand hover:bg-brand-light text-white gap-1.5">
   132→          <Plus className="w-3.5 h-3.5" /> Add Education
   133→        </Button>
   134→      </div>
   135→      <Card className="bg-surface border-stroke overflow-hidden">
   136→        {loading ? (
   137→          <div className="p-6 space-y-3">{[1, 2].map((i) => <Skeleton key={i} className="h-12 w-full bg-dark" />)}</div>
   138→        ) : items.length === 0 ? (
   139→          <div className="p-8 text-center"><p className="text-sm text-muted-text">No education entries yet.</p></div>
   140→        ) : (
   141→          <ScrollArea className="max-h-96">
   142→            <Table>
   143→              <TableHeader>
   144→                <TableRow className="border-stroke hover:bg-transparent">
   145→                  <TableHead className="text-muted-text font-medium">Degree</TableHead>
   146→                  <TableHead className="text-muted-text font-medium hidden md:table-cell">Field</TableHead>
   147→                  <TableHead className="text-muted-text font-medium hidden sm:table-cell">Institution</TableHead>
   148→                  <TableHead className="text-muted-text font-medium hidden lg:table-cell">Year</TableHead>

   162→                        <Button variant="ghost" size="icon" className="h-8 w-8 text-muted-text hover:text-red-500 hover:bg-surface" onClick={() => { setDeleteId(item.id); setDeleteOpen(true); }}><Trash2 className="w-3.5 h-3.5" /></Button>

   430→function SoftSkillsTab() {
   431→  const [items, setItems] = useState<SoftSkill[]>([]);
   432→  const [loading, setLoading] = useState(true);
   433→  const [saving, setSaving] = useState(false);
   434→  const [formOpen, setFormOpen] = useState(false);
   435→  const [deleteOpen, setDeleteOpen] = useState(false);
   436→  const [editingId, setEditingId] = useState<string | null>(null);
   437→  const [deleteId, setDeleteId] = useState<string | null>(null);
   438→  const [form, setForm] = useState(emptySoftSkill);
   439→
   440→  const fetchItems = useCallback(async () => {
   441→    try { const res = await fetch('/api/soft-skills'); if (res.ok) setItems(await res.json()); } catch { /* ignore */ } finally { setLoading(false); }
   442→  }, []);
   443→  useEffect(() => { fetchItems(); }, [fetchItems]);
   444→
   445→  const openCreate = () => { setEditingId(null); setForm(emptySoftSkill); setFormOpen(true); };
   446→  const openEdit = (item: SoftSkill) => { setEditingId(item.id); setForm({ name: item.name, icon: item.icon, displayOrder: item.displayOrder }); setFormOpen(true); };
   447→
   448→  const handleSubmit = async (e: React.FormEvent) => {
   449→    e.preventDefault(); setSaving(true);
   450→    try {
   451→      const res = await fetch('/api/soft-skills', { method: editingId ? 'PUT' : 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(editingId ? { ...form, id: editingId } : form) });
   452→      if (res.ok) { toast.success(editingId ? 'Updated' : 'Created'); setFormOpen(false); fetchItems(); } else toast.error('Failed');
   453→    } catch { toast.error('Failed'); } finally { setSaving(false); }
   454→  };
   455→
   456→  const handleDelete = async () => {
   457→    if (!deleteId) return;
   458→    try { const res = await fetch(`/api/soft-skills?id=${deleteId}`, { method: 'DELETE' }); if (res.ok) { toast.success('Deleted'); setItems((p) => p.filter((i) => i.id !== deleteId)); } else toast.error('Failed'); } catch { toast.error('Failed'); } finally { setDeleteOpen(false); setDeleteId(null); }
   459→  };
   460→
   461→  return (
   462→    <div className="space-y-3">
   463→      <div className="flex justify-end">
   464→        <Button onClick={openCreate} size="sm" className="bg-brand hover:bg-brand-light text-white gap-1.5"><Plus className="w-3.5 h-3.5" /> Add Soft Skill</Button>
   465→      </div>
   466→      <Card className="bg-surface border-stroke overflow-hidden">
   467→        {loading ? (
   468→          <div className="p-6 space-y-3">{[1, 2].map((i) => <Skeleton key={i} className="h-12 w-full bg-dark" />)}</div>
   469→        ) : items.length === 0 ? (
   470→          <div className="p-8 text-center"><p className="text-sm text-muted-text">No soft skills added yet.</p></div>
   471→        ) : (
   472→          <ScrollArea className="max-h-96">
   473→            <Table>
   474→              <TableHeader>
   475→                <TableRow className="border-stroke hover:bg-transparent">
   476→                  <TableHead className="text-muted-text font-medium">Skill</TableHead>
   477→                  <TableHead className="text-muted-text font-medium">Icon</TableHead>
   478→                  <TableHead className="text-muted-text font-medium text-right">Actions</TableHead>
   479→                </TableRow>
   480→              </TableHeader>
   481→              <TableBody>
   482→                {items.map((item) => (
   483→                  <TableRow key={item.id} className="border-stroke hover:bg-dark/50">
   484→                    <TableCell className="font-medium text-white">{item.name}</TableCell>
   485→                    <TableCell className="text-muted-text">{item.icon || '—'}</TableCell>
   486→                    <TableCell className="text-right">
   487→                      <div className="flex items-center justify-end gap-1">
   488→                        <Button variant="ghost" size="icon" className="h-8 w-8 text-muted-text hover:text-white hover:bg-surface" onClick={() => openEdit(item)}><Pencil className="w-3.5 h-3.5" /></Button>
   489→                        <Button variant="ghost" size="icon" className="h-8 w-8 text-muted-text hover:text-red-500 hover:bg-surface" onClick={() => { setDeleteId(item.id); setDeleteOpen(true); }}><Trash2 className="w-3.5 h-3.5" /></Button>
   490→                      </div>
   491→                    </TableCell>
   492→                  </TableRow>
   493→                ))}
   494→              </TableBody>
   495→            </Table>
   496→          </ScrollArea>
   497→        )}
   498→      </Card>
   499→
   500→      <Dialog open={formOpen} onOpenChange={setFormOpen}>
   501→        <DialogContent className="bg-surface border-stroke max-w-md">
   502→          <DialogHeader><DialogTitle className="text-white">{editingId ? 'Edit Soft Skill' : 'Add Soft Skill'}</DialogTitle><DialogDescription className="text-muted-text">Soft skill details.</DialogDescription></DialogHeader>
   503→          <form onSubmit={handleSubmit} className="space-y-4">
   504→            <div className="space-y-2"><Label className="text-sm text-white">Name</Label><Input value={form.name} onChange={(e) => setForm((p) => ({ ...p, name: e.target.value }))} required className="bg-dark border-stroke text-white placeholder:text-muted-text" placeholder="Problem Solving" /></div>
   505→            <div className="space-y-2"><Label className="text-sm text-white">Icon (emoji)</Label><Input value={form.icon} onChange={(e) => setForm((p) => ({ ...p, icon: e.target.value }))} className="bg-dark border-stroke text-white placeholder:text-muted-text" placeholder="🧠" /></div>
   506→            <div className="space-y-2"><Label className="text-sm text-white">Display Order</Label><Input type="number" value={form.displayOrder} onChange={(e) => setForm((p) => ({ ...p, displayOrder: parseInt(e.target.value) || 0 }))} className="bg-dark border-stroke text-white placeholder:text-muted-text" /></div>
